\chapter{Generalităţi}

\section{Module semisimple}

\begin{definition}
Un \(R\)-modul nenul \(S\) se numeşte \emph{simplu} dacă singurele sale submodule sunt
\(0\) şi \(S\).
\end{definition}

\begin{proposition}
Fie \(S\) un \(R\)-modul. Următoarele afirmaţii sunt echivalente:
\begin{enumerate}
  \item \(S\) este modul simplu.
  \item Pentru orice element nenul \(x\in S\) avem \(S = xR\).
  \item \(S \simeq R/I\), unde \(I\) este un ideal drept maximal.
\end{enumerate}
\end{proposition}

\begin{lemma}[Schur]
Fie \(S\) şi \(S'\) două \(R\)-module simple şi \(f\colon S \to S'\) un morfism de \(R\)-module.
Atunci \(f = 0\) sau \(f\) este izomorfism. În particular \(\End_R(S)\) este corp.
\end{lemma}

\begin{definition}
Fie \(M\) un \(R\)-modul şi \((S_i)_{i\in I}\) mulţimea submodulelor simple ale lui \(M\).
Dacă \(M = \sum_{i\in I} S_i\), atunci \(M\) se numeşte \emph{semisimplu}.
\end{definition}

\begin{proposition}
Fie \(M\) un \(R\)-modul semisimplu şi \(N\) un submodul al său.
Atunci există o submulţime \(J \subseteq I\) astfel încât:
\begin{enumerate}
  \item familia \((S_j)_{j\in J}\) este independentă;
  \item \(M = N \oplus \bigl(\bigoplus_{j\in J} S_j\bigr)\).
\end{enumerate}
\end{proposition}

\begin{corollary}
Cu notaţiile de mai sus, pentru modulul semisimplu \(M\) există \(J \subseteq I\) astfel încât
familia \((S_j)_{j\in J}\) este independentă şi
\[
  M = \bigoplus_{i\in I} M_i.
\]
\end{corollary}

\begin{corollary}
Dacă \(M\) este un \(R\)-modul semisimplu şi \(N\) un submodul al său, atunci \(N\) şi
\(M/N\) sunt semisimple.
\end{corollary}

\begin{corollary}
O sumă directă de module semisimple este modul semisimplu.
\end{corollary}

\begin{theorem}
Fie \(M\) un \(R\)-modul. Următoarele afirmaţii sunt echivalente:
\begin{enumerate}
  \item \(M\) este semisimplu;
  \item \(M\) este izomorf cu o sumă directă de module simple;
  \item orice submodul al său este sumand direct în \(M\);
  \item orice şir exact
  \[
    0 \longrightarrow M' \xrightarrow{\,f\,} M \xrightarrow{\,g\,} M'' \longrightarrow 0
  \]
  este scindabil.
\end{enumerate}
\end{theorem}

\begin{definition}
Suma submodulelor simple ale lui \(M\) se numeşte \emph{soclul} lui \(M\) şi se notează
\(\soc(M)\). Dacă \(M\) nu conţine nici un submodul simplu atunci punem \(\soc(M) = 0\).
\end{definition}

\begin{proposition}
Fie \(M\) şi \(N\) două \(R\)-module şi \(f\colon M \to N\) un morfism.
Atunci \(f(\soc(M)) \subseteq \soc(N)\).
\end{proposition}

\begin{proposition}
Fie \(M\) un \(R\)-modul şi \(N\) un submodul al său. Atunci
\[
  \soc(N) = \soc(M) \cap N.
\]
\end{proposition}

\begin{proposition}
Dacă \(M = \bigoplus_{i\in I} M_i\), atunci
\[
  \soc(M) = \bigoplus_{i\in I} \soc(M_i).
\]
\end{proposition}

\begin{proposition}
Fie \(R\) un inel. Atunci \(\soc(R_R)\) este un ideal bilateral al lui \(R\).
\end{proposition}



\section{Module noetheriene (artiniene) şi inele noetheriene (artiniene)}

\begin{definition}
Fie \(R\) un inel şi \(M\) un \(R\)-modul drept. Spunem că \(M\) satisface
\emph{condiţia maximală} (resp. \emph{minimală}) dacă orice mulţime nevidă de
submodule ale lui \(M\), ordonată prin incluziune, admite un element maximal
(resp. minimal).

Spunem că \(M\) satisface \emph{condiţia lanţurilor ascendente} (resp.
\emph{descendente}) dacă orice şir (lanţ) ascendent de submodule ale lui \(M\)
\[
  M_1 \subseteq M_2 \subseteq \cdots \subseteq M_i \subseteq \cdots
\]
(resp. orice şir descendent
\[
  M_1 \supseteq M_2 \supseteq \cdots \supseteq M_i \supseteq \cdots
\])
este staţionar, adică există \(n \ge 1\) astfel încât
\(M_n = M_{n+1} = \cdots\).
\end{definition}

\begin{proposition}
Fie \(M\) un \(R\)-modul. Următoarele afirmaţii sunt echivalente:
\begin{enumerate}
  \item \(M\) satisface condiţia maximală (minimală);
  \item \(M\) satisface condiţia lanţurilor ascendente (descendente).
\end{enumerate}
\end{proposition}

\begin{definition}
Un \(R\)-modul \(M\) se numeşte \emph{noetherian} (resp. \emph{artinian}) dacă
satisface condiţia maximală (resp. minimală). Inelul \(R\) se numeşte
noetherian (resp. artinian) la dreapta dacă \(R_R\) este noetherian
(resp. artinian).
\end{definition}

\begin{example}
\leavevmode
\begin{enumerate}
  \item \(\mathbb{Z}\) este inel noetherian dar nu este artinian.
  \item Orice grup finit este \(\mathbb{Z}\)-modul noetherian şi artinian.
  \item Orice inel finit este noetherian şi artinian.
  \item \(\mathbb{Z}[X_1,X_2,\ldots,X_n,\ldots]\) nu este nici noetherian,
        nici artinian:
        \[
          (X_1) \subsetneq (X_1,X_2) \subsetneq \cdots \subsetneq
          (X_1,\ldots,X_n) \subsetneq \cdots
        \]
        \[
          (X_1) \supsetneq (X_1^2) \supsetneq \cdots \supsetneq
          (X_1^k) \supsetneq \cdots
        \]
  \item \(\mathbb{Z}_{p^\infty}\) este \(\mathbb{Z}\)-modul artinian dar nu
        este noetherian.
\end{enumerate}
\end{example}

\begin{proposition}
Fie \(N, P\) două submodule ale lui \(M\) astfel încât \(M = N + P\).
Atunci \(M\) este noetherian (artinian) dacă şi numai dacă \(N\) şi \(P\) sunt
noetheriene (artiniene).
\end{proposition}

\begin{proposition}
Pentru un \(R\)-modul \(M\) următoarele afirmaţii sunt echivalente:
\begin{enumerate}
  \item \(M\) este noetherian;
  \item orice submodul al lui \(M\) este finit generat.
\end{enumerate}
\end{proposition}

\begin{proposition}
Pentru un \(R\)-modul \(M\) următoarele afirmaţii sunt echivalente:
\begin{enumerate}
  \item \(M\) este artinian;
  \item oricare ar fi familia \((X_i)_{i\in I}\) de submodule ale lui \(M\),
        există \(J \subseteq I\), \(J\) finită, astfel încât
        \[
          \bigcap_{i\in I} X_i = \bigcap_{j\in J} X_j.
        \]
\end{enumerate}
\end{proposition}

\section{Module de lungime finită}

\begin{definition}
Fie \(M\) un \(R\)-modul drept nenul. Se numeşte \emph{şir de compoziţie} sau
\emph{şir Jordan--Hölder} al lui \(M\) un lanţ finit strict ascendent de
submodule
\[
  0 = X_0 \subset X_1 \subset \cdots \subset X_n = M
\]
astfel încât \(X_{i+1}/X_i\) este modul simplu pentru \(0 \le i \le n-1\).
Numărul \(n\) se numeşte \emph{lungimea şirului}, iar modulele
\(X_{i+1}/X_i\) se numesc \emph{factorii şirului}.
\end{definition}

\begin{proposition}
Fie \(M\) un \(R\)-modul. Următoarele afirmaţii sunt echivalente:
\begin{enumerate}
  \item \(M\) are un şir de compoziţie;
  \item \(M\) este noetherian şi artinian.
\end{enumerate}
\end{proposition}

\begin{proposition}
Fie
\[
  0 \longrightarrow M' \longrightarrow M \longrightarrow M'' \longrightarrow 0
\]
un şir exact de \(R\)-module drepte. Atunci \(M\) admite un şir de compoziţie
dacă şi numai dacă \(M'\) şi \(M''\) admit un şir de compoziţie.
\end{proposition}

Dacă
\[
  0 = M_0 \subseteq M_1 \subseteq \cdots \subseteq M_n = M,\qquad
  0 = N_0 \subseteq N_1 \subseteq \cdots \subseteq N_p = M
\]
sunt două şiruri de compoziţie ale lui \(M\), vom spune că ele sunt
\emph{echivalente} dacă \(n=p\) şi există o bijecţie
\(\sigma : \{0,\ldots,n-1\} \to \{0,\ldots,n-1\}\) astfel încât
\[
  M_{i+1}/M_i \cong M_{\sigma(i)+1}/M_{\sigma(i)}
  \quad (0 \le i \le n-1).
\]

\begin{theorem}[Jordan--Hölder]
Dacă un \(R\)-modul \(M\) are două şiruri de compoziţie
\[
  0 = M_0 \subseteq M_1 \subseteq \cdots \subseteq M_n = M,\qquad
  0 = N_0 \subseteq N_1 \subseteq \cdots \subseteq N_p = M,
\]
atunci aceste două şiruri sunt echivalente.
\end{theorem}

\begin{definition}
Un \(R\)-modul \(M\) care admite un şir de compoziţie se numeşte
\emph{modul de lungime finită}. Lungimea şirurilor de compoziţie se
numeşte \emph{lungimea} lui \(M\) şi se notează \(l(M)\). Dacă \(M\)
nu admite nici un şir de com\-po\-zi\-ţie, atunci spunem că \(M\) este
\emph{de lungime infinită} şi scriem \(l(M)=\infty\).
\end{definition}

\begin{proposition}
Fie
\[
  0 \longrightarrow M' \longrightarrow M \longrightarrow M'' \longrightarrow 0
\]
un şir exact de \(R\)-module de lungime finită. Atunci
\[
  l(M) = l(M') + l(M'').
\]
\end{proposition}

\begin{corollary}
Fie \(M\) un \(R\)-modul de lungime finită şi \(N,L\) două submodule ale
sale. Atunci:
\begin{enumerate}
  \item \(l(M) = l(N) + l(M/N)\);
  \item \(l(N+L) + l(N\cap L) = l(N) + l(L)\).
\end{enumerate}
\end{corollary}

\begin{corollary}
Fie \(M\) un \(R\)-modul de lungime finită şi \(M_1,M_2,\ldots,M_n\)
submodule ale sale astfel încât
\[
  M = M_1 \oplus M_2 \oplus \cdots \oplus M_n.
\]
Atunci
\[
  l(M) = \sum_{i=1}^{n} l(M_i).
\]
\end{corollary}

\section{Radicalul Jacobson}
\subsection*{Radicalul Jacobson al unui modul}

% \medskip
% \noindent\textbf{Radicalul Jacobson al unui modul.}

\begin{definition}
Fie \(M\) un \(R\)-modul. Intersecţia tuturor submodulelor maximale ale
lui \(M\) se numeşte \emph{radicalul Jacobson} al modulului \(M\) şi se
notează \(\Rad(M)\). Dacă \(M\) nu are nici un submodul maximal, atunci
prin convenţie punem \(\Rad(M)=M\).
\end{definition}

\begin{remark}
Dacă \(M\) este un \(R\)-modul finit generat, atunci \(\Rad(M)\ne M\).
\end{remark}

\begin{proposition}
Fie \(M\) un \(R\)-modul. Atunci
\[
  \Rad(M)
  = \bigcap_{\substack{f : M \to S\\ S\ \text{simplu}}} \ker(f)
  = \bigcap_{\substack{f : M \to X\\ X\ \text{semisimplu}}} \ker(f).
\]
\end{proposition}

\begin{proposition}
Fie \(f : M \to N\) un morfism de \(R\)-module. Atunci
\(f(\Rad(M)) \subseteq \Rad(N)\). Dacă, în plus, \(f\) este epimorfism şi
\(\ker(f) \subseteq \Rad(M)\), atunci \(f(\Rad(M)) = \Rad(N)\).
\end{proposition}

\begin{corollary}
Pentru orice \(R\)-modul \(M\) are loc egalitatea
\(\Rad(M/\Rad(M)) = 0\).
\end{corollary}

\begin{corollary}
Dacă \(M\) este un \(R\)-modul semisimplu, atunci \(\Rad(M)=0\).
\end{corollary}

\begin{corollary}
Dacă \(M = \bigoplus_{i\in I} M_i\), atunci
\[
  \Rad(M) = \bigoplus_{i\in I} \Rad(M_i).
\]
\end{corollary}

\begin{proposition}
Fie \(M\) un \(R\)-modul astfel încât \(\Rad(M)\ne M\).
Atunci
\[
  \Rad(M)
  = \bigcap \{\,L \le M \mid L \text{ este submodul superfluu}\,\}.
\]
\end{proposition}

\begin{proposition}[Lema lui Nakayama]
Fie \(M\) un \(R\)-modul finit generat şi \(N\) un submodul al său.
Dacă \(N + \Rad(M) = M\), atunci \(N = M\). (Adică \(\Rad(M)\) este
cel mai mare submodul superfluu al lui \(M\).)
\end{proposition}

\subsection*{Radicalul Jacobson al unui inel}

Fie \(R\) un inel. Considerăm idealul stâng
\(\Rad({}_R R)\) ca intersecţie a idealelor stângi maximale ale lui \(R\)
şi \(\Rad(R_R)\) ca intersecţie a idealelor drepte maximale ale lui \(R\).

\begin{proposition}
\leavevmode
\begin{enumerate}
  \item \(\Rad(R_R)\) este un ideal bilateral.
  \item \(\Rad(R_R) = \{\,r \in R \mid 1-ar \in U(R)\ \text{pentru orice } a \in R\,\}\).
  \item \(\Rad(R_R) = \Rad({}_R R)\).
\end{enumerate}
\end{proposition}

\begin{definition}
Idealul bilateral \(\Rad(R_R) = \Rad({}_R R)\) se numeşte
\emph{radicalul Jacobson} al inelului \(R\) şi se notează \(\Rad(R)\).
\end{definition}

\begin{proposition}
\leavevmode
\begin{enumerate}
  \item Dacă \(J\) este un ideal stâng (resp.\ drept sau bilateral) cu
        proprietatea că \(1-x\) este inversabil pentru orice \(x \in J\),
        atunci \(J \subseteq \Rad(R)\).
  \item Dacă \(J\) este un nilideal stâng (resp.\ drept sau bilateral),
        atunci \(J \subseteq \Rad(R)\).
\end{enumerate}
\end{proposition}

\begin{proposition}
Fie \(\varphi : R \to S\) un morfism surjectiv de inele.
Atunci \(\varphi(\Rad(R)) \subseteq \Rad(S)\). Dacă \(\ker(\varphi)
\subseteq \Rad(R)\), atunci \(\varphi(\Rad(R)) = \Rad(S)\).
\end{proposition}

\begin{proposition}
Dacă \((R_i)_{i\in I}\) este o familie de inele, atunci
\[
  \Rad\Bigl(\prod_{i\in I} R_i\Bigr) = \prod_{i\in I} \Rad(R_i).
\]
\end{proposition}

\begin{proposition}
Fie \(M\) un \(R\)-modul. Atunci \(M\,\Rad(R) \subseteq \Rad(M)\).
\end{proposition}

\begin{theorem}
Dacă \(R\) este un inel artinian, atunci \(\Rad(R)\) este nilpotent.
\end{theorem}

\section{Inele semisimple}

\begin{theorem}
Pentru un inel \(R\) următoarele afirmaţii sunt echivalente:
\begin{enumerate}
  \item orice \(R\)-modul drept nenul este semisimplu;
  \item \(R\) este \(R\)-modul drept semisimplu;
  \item \(R\) este artinian şi \(\Rad(R)=0\).
\end{enumerate}
\end{theorem}

\begin{definition}
Un inel \(R\) care satisface una din condiţiile de mai sus se numeşte
\emph{inel semisimplu}.
\end{definition}

\begin{proposition}
Fie \(R\) un inel semisimplu şi \(M\) un \(R\)-modul nenul.
Următoarele afirmaţii sunt echivalente:
\begin{enumerate}
  \item \(M\) este de lungime finită;
  \item \(M\) este noetherian;
  \item \(M\) este artinian.
\end{enumerate}
\end{proposition}

\begin{theorem}
Fie \(R\) un inel artinian la dreapta şi \(M\) un \(R\)-modul nenul.
Următoarele afirmaţii sunt echivalente:
\begin{enumerate}
  \item \(M\) este de lungime finită;
  \item \(M\) este noetherian;
  \item \(M\) este artinian.
\end{enumerate}
\end{theorem}

\begin{corollary}[Hopkins]
Un inel artinian la dreapta (respectiv la stânga) este noetherian la
dreapta (respectiv la stânga).
\end{corollary}






\chapter{Submodule esenţiale}

\begin{definition}\label{def:submodul-esential}
Fie \(M\) un \(R\)-modul drept. Un submodul \(N\) al lui \(M\) se numeşte
\emph{esenţial} (sau spunem că \(M\) este o extensie esenţială a lui \(N\))
dacă \(N \cap N' \neq 0\) pentru orice submodul nenul \(N'\) al lui \(M\).
În acest caz vom folosi notaţia \(N \ess M_R\).

Un monomorfism de \(R\)-module la dreapta \(f : M \to N\) se numeşte
\emph{esenţial} dacă \(\operatorname{Im} f\) este submodul esenţial în
\(N\) (adică \(\operatorname{Im} f \ess N_R\)).
\end{definition}


\begin{example*}
\leavevmode
\begin{enumerate}
  \item \(n\mathbb{Z} \ess \mathbb{Z}_\mathbb{Z}\) pentru orice \(n \geq 1\).
  \item Orice submodul al lui \(\mathbb{Z}_{p^\infty}\) este esenţial.
\end{enumerate}
\end{example*}

  
\begin{remark}
Fie \(M\) un \(R\)-modul drept şi \(N\) un submodul al lui \(M\).
Atunci \(N \ess M_R\) dacă şi numai dacă pentru orice \(x \in M\),
\(x \neq 0\), există \(r \in R\) astfel încât \(xr \in N \setminus \{0\}\).
\end{remark}

\begin{proof}
„\(\Rightarrow\)” Fie \(x \in M \setminus \{0\}\).
Cum \(0 \neq xR \subseteq M_R\) şi \(N \ess M_R\), rezultă că
\(xR \cap N \neq 0\), deci există \(xr \in xR \cap N \setminus \{0\}\).

„\(\Leftarrow\)” Fie \(N' \leq M_R\), \(N' \neq 0\).
Pentru \(x \in N' \setminus \{0\}\) există \(r \in R\) astfel încât
\(xr \in N \setminus \{0\}\), deci \(N \cap N' \neq 0\).
\end{proof}

\begin{definition}
Un monomorfism de \(R\)-module la dreapta \(f : N_R \to M_R\) se numeşte
esenţial dacă \(\operatorname{Im} f \ess M_R\).
Se observă imediat că dacă \(N\) este un submodul al lui \(M\) atunci
incluziunea canonică \(i_N : N \to M\) este monomorfism esenţial
dacă şi numai dacă \(N \ess M_R\).
\end{definition}

\begin{proposition}
Un monomorfism \(f : N_R \to M_R\) este esenţial dacă şi numai dacă
pentru orice \(R\)-modul drept \(M'\) şi orice \(g \in \Hom(M,M')\),
faptul că \(g \circ f\) este monomorfism implică \(g\) monomorfism.
\end{proposition}

\begin{proof}
\mbox{}\\[0.5em]
„\(\Rightarrow\)” Fie \(g\) ca în enunţ astfel încât \(g \circ f\) este
monomorfism. Presupunem \(g \neq 0\).
Fie \(x \in \Ker g \cap \operatorname{Im} f \setminus \{0\}\).
Există \(x' \in N\) astfel încât \(x = f(x')\) şi \(g(x) = 0\), de unde
\(g(f(x')) = 0\). Cum \(g \circ f\) este monomorfism, rezultă \(x' = 0\) şi
deci \(x = 0\), contradicţie.

„\(\Leftarrow\)” Dacă \(f\) nu este monomorfism esenţial atunci există
\(N' \leq M_R\), \(N' \neq 0\), astfel încât \(N' \cap \operatorname{Im} f = 0\).
Considerăm proiecţia canonică \(\pi_{N'} : M \to M/N'\).
Dacă \(x \in \Ker(\pi_{N'} \circ f)\), atunci \(f(x) \in N'\), deci \(f(x) = 0\),
adică \(x = 0\). Obţinem astfel că \(\pi_{N'} \circ f\) este injectiv, de unde
rezultă că \(\pi_{N'}\) este injectiv, ceea ce implică \(N' = 0\),
contradicţie.
\end{proof}

\begin{corollary}
Fie \(M\) un \(R\)-modul la dreapta şi \(N \leq M_R\).
Atunci următoarele afirmaţii sunt echivalente:
\begin{enumerate}
  \item \(N \ess M_R\);
  \item incluziunea \(i_N : N \to M\) este monomorfism esenţial;
  \item pentru orice \(f \in \Hom(M,M')\) cu \(M'\) \(R\)-modul arbitrar,
        faptul că \(f \circ i_N\) este monomorfism implică \(f\) monomorfism.
\end{enumerate}
\end{corollary}

\begin{proposition}
Fie \(f : N_R \to M_R\) şi \(g : M_R \to P_R\) două monomorfisme.
Atunci \(g \circ f\) este esenţial dacă şi numai dacă \(g\) şi \(f\) sunt
esenţiale.
\end{proposition}

„\(\Leftarrow\)” Fie \(z \in P \setminus \{0\}\). Cum \(g\) este

\(y \in M \setminus \{0\}\) astfel încât \(zr = g(y)\).


Cum \(f\) este esenţial, există \(r' \in R\) astfel încât \(y r' \in \mathrm{Im}\, f \setminus \{0\}\).
De aici există \(x \in N \setminus \{0\}\) astfel încât \(y r' = f(x)\).
Dar \(zr' = g(y) r' = g(y r') = g(f(x))\). Dacă \(zr' = 0\), atunci \(g(f(x)) = 0\)
şi deci \(x = 0\), contradicţie. Obţinem astfel că \(zr' \in \mathrm{Im}(g \circ f)\)
şi că \(zr' \neq 0\), ceea ce ne arată că \(g \circ f\) este esenţial.

„\(\Rightarrow\)” Fie \(y \in M \setminus \{0\}\).
Cum \(g\) este monomorfism, \(g(y) \neq 0\).
Deci există \(r \in R\) astfel încât \(g(yr) \in \mathrm{Im}\, g \setminus \{0\}\)
şi \(g(yr) \neq 0\). Rezultă că există \(x \in N \setminus \{0\}\) astfel încât
\(g(yr) = g(f(x))\) de unde \(yr = f(x) \in \mathrm{Im}\, f\), ceea ce ne arată
că \(f\) este monomorfism esenţial.

Dacă \(z \in P \setminus \{0\}\) există \(r \in R\) astfel încât \(zr \in \mathrm{Im}(g \circ f)\)
şi \(zr \neq 0\). Cum \(\mathrm{Im}(g \circ f) \subseteq \mathrm{Im} g\),
rezultă că \(zr \in \mathrm{Im} g\), şi deci \(g\) este monomorfism esenţial.

\end{proof}


\begin{proposition}
Fie \(M\) un \(R\)-modul la dreapta şi \(L_1, L_2, \ldots, L_n\) submodule ale lui \(M\).
Atunci:

1) \(\displaystyle \bigcap_{i=1}^n L_i\) este esenţial în \(M\) dacă şi numai dacă
\(L_i\) este esenţial în \(M\) pentru orice \(i = 1,\ldots,n\).

2) Dacă \(L_1 \subseteq L_2\) şi \(L_1\) este esenţial în \(M\), atunci
\(L_2\) este esenţial în \(M\).

Demonstraţia este evidentă.
\end{proposition}



\begin{proposition}
Fie \(K\) şi \(L\) două submodule ale lui \(M\).
1) Dacă \(K \subseteq L \subseteq M\), atunci \(K \, \ess \, M\) dacă şi numai dacă
\(K \, \ess \, L\) şi \(L \, \ess \, M\).

2) Dacă \(h : K_R \to M_R\) este morfism de module şi \(L \, \ess \, M\),
atunci \(h^{-1}(L) \, \ess \, K\).

3) Dacă \(L_1, L_2 \leq M_R\) şi \(K_1 \, \ess \, L_1\), \(K_2 \, \ess \, L_2\),
atunci \(K_1 \cap K_2 \, \ess \, L_1 \cap L_2\).
\end{proposition}

\begin{proof}
\mbox{}\\[0.1em]
1) Se aplică 1.5 şi 1.6.

2) Fie \(U\) submodul nenul al lui \(K\).
(i) Dacă \(h(U) = 0\), atunci \(U \subseteq \ker h \subseteq h^{-1}(L)\),
ceea ce implică \(U \cap h^{-1}(L) \neq 0\).
(ii) Dacă \(h(U) \neq 0\), atunci \(h(U) \cap L \neq 0\) şi deci există
\(u \in U\) astfel încât \(h(u) \in L\), \(h(u) \neq 0\), de unde
\(u \in U \cap h^{-1}(L)\) şi \(u \neq 0\).
Din (i) şi (ii) rezultă că \(h^{-1}(L) \, \ess \, K\).

3) Dacă \(0 \neq X \leq L_1 \cap L_2\) atunci \(X \subseteq L_1\) ceea ce implică
\(0 \neq X \cap K_1 \leq L_1\). Dar cum \(X \subseteq L_2\),
rezultă \(0 \neq (X \cap K_1) \cap L_2 = X \cap (K_1 \cap K_2)\),
şi deci \(K_1 \cap K_2 \, \ess \, L_1 \cap L_2\).
\end{proof}


\begin{proposition}
Fie \((K_\lambda)_{\lambda \in \Lambda}\), \((L_\lambda)_{\lambda \in \Lambda}\)
două familii de submodule ale lui \(M\).
Dacă \((K_\lambda)_{\lambda \in \Lambda}\) este familie independentă în \(M\)
şi \(K_\lambda \, \ess \, L_\lambda\) pentru orice \(\lambda \in \Lambda\),
atunci \((L_\lambda)_{\lambda \in \Lambda}\) este familie independentă în \(M\)
şi \((\oplus_{\lambda \in \Lambda} K_\lambda) \, \ess \,
(\oplus_{\lambda \in \Lambda} L_\lambda)\).
\end{proposition}

\begin{proof}
\mbox{}\\[0.5em]
Fie \(K_1 \, \ess \, L_1\), \(K_2 \, \ess \, L_2\) astfel încât
\(K_1 \cap K_2 = 0\). Din 1.8(3) rezultă că \(0 \, \ess \, L_1 \cap L_2\),
adică \(L_1 \cap L_2 = 0\).

Fie proiecţiile canonice
\(\pi_1 : L_1 \oplus L_2 \to L_1\),
\(\pi_2 : L_1 \oplus L_2 \to L_2\).
Cum \(K_1 \, \ess \, L_1\), \(K_2 \, \ess \, L_2\) rezultă că

\[
\pi_1^{-1}(K_1)
  = K_1 \oplus 0 \, \ess \, L_1 \oplus L_2,
\]
şi
\[
\pi_2^{-1}(K_2)
  = 0 \oplus K_2 \, \ess \, L_1 \oplus L_2.
\]

Deci
\[
K_1 \oplus K_2
  = (\pi_1^{-1}(K_1)) \cap (\pi_2^{-1}(K_2))
  \, \ess \, L_1 \oplus L_2.
\]

Prin inducţie se obţine afirmaţia pentru mulţimi finite.
În cazul general, fie \(0 \neq m \in \oplus_{\lambda \in \Lambda} L_\lambda\).
Atunci există o mulţime finită \(\Lambda_0 \subseteq \Lambda\)
cu \(m \in \oplus_{\lambda \in \Lambda_0} L_\lambda\).
Cum \((\oplus_{\lambda \in \Lambda_0} K_\lambda) \, \ess \,
(\oplus_{\lambda \in \Lambda_0} L_\lambda)\),
există \(r \in R\) astfel încât
\(rm \in (\oplus_{\lambda \in \Lambda_0} K_\lambda) \setminus \{0\}
\subseteq (\oplus_{\lambda \in \Lambda} K_\lambda) \setminus \{0\}\).
Rezultă că
\((\oplus_{\lambda \in \Lambda} K_\lambda) \, \ess \,
(\oplus_{\lambda \in \Lambda} L_\lambda)\).
\end{proof}


\begin{proposition}
Fie \(N\) un submodul al lui \(M\).
Atunci există un submodul \(Q\), \(N \subseteq Q \subseteq M\),
astfel încât \(Q\) este o extensie esenţială maximală a lui \(N\)
conţinută în \(M\).
\end{proposition}

\begin{proof}
\mbox{}\\[0.5em]
Fie \(\mathfrak{S} = \{ L \leq M ; \; N \subseteq L \subseteq M, \; N \, \ulcorner \, L \}\),
\(\mathfrak{S}\) cu relaţia de ordine incluziunea.
\(\mathfrak{S} \neq \varnothing\) deoarece \(N \in \mathfrak{S}\).
Fie \((L_\lambda)_{\lambda \in \Lambda}\) o familie total ordonată de elemente din
\(\mathfrak{S}\) şi
\[
L := \bigcup_{\lambda \in \Lambda} L_\lambda.
\]
Evident \(L \leq M_R\).

Fie \(x \in L \setminus \{0\}\).
Atunci există \(\lambda_0 \in \Lambda\) cu \(x \in L_{\lambda_0}\).
Cum \(N\) este esenţial în \(L_{\lambda_0}\), rezultă că există \(r \in R\) astfel
încât \(xr \in N\) şi \(xr \neq 0\), de unde obţinem că \(L\) este extensie esenţială
a lui \(N\).
Deci \(\mathfrak{S}\) este inductivă şi, conform lemei lui Zorn, \(\mathfrak{S}\) admite
un element maximal \(Q\) care satisface condiţiile cerute.
\end{proof}

\begin{definition}
Fie \(M\) un \(R\)-modul la dreapta şi \(N \leq M_R\).
Un submodul \(K \leq M_R\) se numeşte \emph{complement} al lui \(N\) în \(M\)
dacă \(K\) este un submodul maximal al lui \(M\) cu proprietatea că
\(K \cap N = 0\).
Un submodul \(K \leq M_R\) se numeşte \emph{submodul complement al lui \(M\)}
dacă există \(N \leq M_R\) astfel încât \(K\) este complement al lui \(N\) în \(M\).
\end{definition}

\begin{remark}
Mulţimea
\[
\widetilde{\mathfrak{S}}
  = \{\,L \leq M_R \mid N \cap L = 0\,\}
\]
este inductivă şi, aplicând lema lui Zorn, rezultă că există un complement al lui
\(N\) în \(M\).
În particular, \(0\) şi \(M\) sunt submodule complement ale lui \(M\).
\end{remark}

\begin{proposition}
Fie \(M_R\), \(N \leq M_R\) şi \(K \leq M_R\), \(K\) un complement al lui \(N\) în \(M\).
Există un complement \(Q\) al lui \(K\) în \(M\) astfel încât \(N \subseteq Q\).
Mai mult, \(Q\) este o extensie esenţială maximală a lui \(N\) în \(M\).
\end{proposition}

\begin{proof}
Se observă uşor că mulţimea
\[
\widetilde{\mathfrak{S}}
  = \{\,L \leq M_R \mid K \cap L = 0,\; N \subseteq L\,\}
\]
este inductivă şi lema lui Zorn asigură existenţa lui \(Q\).

Fie \(L\) un submodul nenul al lui \(Q\) astfel încât \(L \cap N = 0\).
Fie \(K_1 = L + K\).
Este clar că \(K \subseteq K_1\).
Dacă \(x \in N \cap (L + K)\), atunci \(x = y + z\) cu \(y \in L\), \(z \in K\).
Dar \(z = x - y \in Q\).
Cum \(Q \cap K = 0\), rezultă \(z = 0\) şi deci \(x = y\).
Din egalitatea \(L \cap N = 0\) deducem \(x = y = 0\) şi deci
\(N \cap (L + K) = 0\), ceea ce contrazice faptul că \(K\) este un complement
al lui \(N\) în \(M\).
Obţinem \(L \cap N \neq 0\) pentru orice \(0 \neq L \leq Q\), de unde
\(Q\) este extensie esenţială a lui \(N\).

Presupunem că există \(Q' \leq M_R\) cu \(N \ess Q'\) şi \(Q \subsetneq Q'\).
Cum \(Q'\) este complement al lui \(K\), rezultă \(Q' \cap K \neq 0\).
Dar \(N \cap (Q' \cap K) = 0\) şi
\(0 \neq Q' \cap K \leq Q'\), contradicţie cu \(N \ess Q'\).
Rezultă că \(Q\) este extensie esenţială maximală a lui \(N\) în \(M\).
\end{proof}

\begin{definition}
Un submodul \(N\) al lui \(M_R\) se numeşte \emph{închis} dacă \(N\) nu are
nicio extensie esenţială în \(M\) proprie (diferită de \(N\)).
\end{definition}

\begin{corollary}
Fie \(M_R\) un \(R\)-modul.
Submodulele complement ale lui \(M\) coincid cu submodulele închise ale lui \(M\).
\end{corollary}

\begin{proof}
Din 1.13 rezultă imediat că orice submodul închis al lui \(M\) este un submodul
complement al lui \(M\).

Invers, fie \(K\) un submodul complement al lui \(M_R\).
Rezultă că există \(N \leq M_R\) astfel încât \(K\) este un complement al lui \(N\)
în \(M\).
Presupunem că \(K\) are o extensie esenţială în \(M\) proprie, adică există
\(K' \leq M_R\) cu \(K \ess K'\) şi \(K \subsetneq K'\).
Atunci \(K' \cap N \neq 0\), din maximalitatea lui \(K\), iar cum \(K \ess K'\),
rezultă că
\[
K \cap K' \cap N \neq 0,
\]
contradicţie.
\end{proof}


\begin{corollary}
Fie \(N\) un submodul al lui \(M_R\).
Dacă \(K\) este un complement al lui \(N\) în \(M\), atunci:
\begin{enumerate}
  \item \((N + K) \ess M_R\).
  \item Morfismul canonic \(\pi_K \circ i_N : N \to M/K\) este monomorfism esenţial.
\end{enumerate}
\end{corollary}

\begin{proof}
(1) Fie \(x \in M \setminus \{0\}\).
Dacă \(x \notin K\), atunci \(K + Rx \neq K\) şi deci
\(N \cap (K + Rx) \neq 0\).
Fie \(y \in N \cap (K + Rx)\), \(y \neq 0\).
Există \(z \in K\), \(r \in R\) cu \(y = z + rx\).
Dacă \(rx = 0\), atunci \(y = z\) şi cum \(N \cap K = 0\) rezultă \(y = 0\),
contradicţie.
Deci \(rx \neq 0\) şi, cum \(rx = y - z\), obţinem \(rx \in N + K\), ceea ce
ne arată că \((N + K) \ess M_R\).

(2) \(\mathrm{Im}(\pi_K \circ i_N) = (N + K)/K\).
Fie \(L/K\) un submodul nenul al lui \(M/K\).
Atunci
\[
\frac{N + K}{K} \cap \frac{L}{K}
   = \frac{(N + K) \cap L}{K}
   = \frac{N \cap L + K}{K}.
\]
Cum \(K\) este un complement al lui \(N\), rezultă că \(N \cap L \neq 0\) şi deci
\[
\frac{N \cap L + K}{K} \neq 0,
\]
ceea ce ne arată că \(\pi_K \circ i_N\) este monomorfism esenţial.
\end{proof}




\chapter{Module injective}
\section{Module injective}

Fie \(Q\) şi \(M\) două \(R\)-module drepte.
\(Q\) se numeşte \emph{\(M\)-injectiv} dacă pentru orice monomorfism
\(u : M' \to M\) şi orice morfism \(f : M' \to Q\), există
\(g : M \to Q\) astfel încât \(g \circ u = f\), adică diagrama

\[
\begin{tikzcd}
0 \arrow[r] & M' \arrow[r,"u"] \arrow[d,"f"'] & M \arrow[dl,dashed,"g"] \\
            & Q &
\end{tikzcd}
\]

este comutativă.

Această proprietate este echivalentă cu condiţia ca aplicaţia
\[
\Hom(u,Q) : \Hom(M,Q) \longrightarrow \Hom(M',Q)
\]
să fie surjectivă pentru orice monomorfism \(u : M' \to M\).
Cum functorul \(\Hom(-,Q)\) este exact la stânga, rezultă că \(Q\) este
\(M\)-injectiv dacă şi numai dacă \(\Hom(-,Q)\) este exact în raport cu orice
şir exact de forma
\[
0 \longrightarrow M' \longrightarrow M \longrightarrow M'' \longrightarrow 0.
\]

\(R\)-modulul \(Q\) se numeşte \emph{quasi-injectiv} (sau \emph{self-injectiv})
dacă este \(Q\)-injectiv.
Dacă \(Q\) este \(M\)-injectiv pentru orice \(R\)-modul \(M\), atunci
\(Q\) se numeşte \emph{injectiv}.

2.1.1 Propoziţie

\begin{proposition}
Fie \(Q\) şi \(M\) două \(R\)-module.
Următoarele afirmaţii sunt echivalente:
\begin{enumerate}
  \item \(Q\) este \(M\)-injectiv.
  \item Pentru orice submodul \(N\) al lui \(M\) şi orice morfism
        \(f : N \to Q\), există \(g : M \to Q\) astfel încât
        \(g_{\mid N} = f\).
  \item Pentru orice submodul esenţial \(N\) al lui \(M\) şi orice morfism
        \(f : N \to Q\), există \(g : M \to Q\) astfel încât
        \(g_{\mid N} = f\).
\end{enumerate}
\end{proposition}

\begin{proof}
Implicaţiile \((1) \Rightarrow (2)\) şi \((2) \Rightarrow (3)\) sunt evidente.

\medskip
\noindent\((2) \Rightarrow (1)\).
Fie \(M'_R\), \(0 \longrightarrow M' \xrightarrow{u} M\) şi
\(f : M' \to Q\).
Atunci \(u(M') \leq M\).
Considerăm \(i : u(M') \to M\) injecţia canonică şi
\(\bar{u} : M' \to u(M')\) izomorfismul indus de \(u\).
Există \(g : M \to Q\) astfel încât \(g \circ i = f \circ \bar{u}^{-1}\).
Atunci
\[
g \circ i \circ \bar{u} = f
\quad\text{şi deci}\quad
g \circ u = f.
\]

Diagrama corespunzătoare este

\[
\begin{tikzcd}
0 \arrow[r] &
M' \arrow[r,shift left=0.35ex,"\bar{u}"]
   \arrow[d,"f"'] &
u(M') \arrow[r,hook,"i"] &
M \arrow[dll,dashed,"g"] \\
& Q &
\end{tikzcd}
\]



\noindent\((3) \Rightarrow (2)\).
Fie \(N\) un submodul al lui \(M\) şi \(K\) un complement al lui \(N\) în \(M\).
Atunci \((N \oplus K) \ess M\).
Fie \(h : N \oplus K \to Q\), definit prin \(h(n+k) = f(n)\) pentru orice
\(n \in N\), \(k \in K\).
Cum \(N \cap K = 0\), aplicaţia \(h\) este bine definită.
Există \(g : M \to Q\) astfel încât \(g_{\mid N \oplus K} = h\) şi deci
\(g_{\mid N} = h_{\mid N} = f\).

Aceasta se poate reprezenta prin diagrama
\[
\begin{tikzcd}
0 \arrow[r] &
N \arrow[r,hook] \arrow[d,"f"'] &
N \oplus K \arrow[r,hook] \arrow[dl,bend right=5,"h"] &
M \arrow[dll,dashed,bend left=10,"g"] \\
& Q &
\end{tikzcd}
\]


\end{proof}


\begin{proposition}
Fie \((M_\alpha)_{\alpha \in \Lambda}\) o familie de \(R\)-module şi
\(M\) un \(R\)-modul.
Atunci \(\displaystyle \prod_{\alpha \in \Lambda} M_\alpha\) este
\(M\)-injectiv dacă şi numai dacă \(M_\alpha\) este \(M\)-injectiv
pentru orice \(\alpha \in \Lambda\).
\end{proposition}

\begin{proof}
Fie \(N\) un submodul al lui \(M\).
Notăm \(P = \displaystyle\prod_{\alpha \in \Lambda} M_\alpha\) şi
\(\pi_\alpha : P \to M_\alpha\) proiecţiile canonice, pentru orice
\(\alpha \in \Lambda\).

\medskip
\noindent\("\Leftarrow"\) Considerând un morfism \(f : N \to P\), avem că
morfismele \(\pi_\alpha \circ f : N \to M_\alpha\) pot fi extinse la
\(g_\alpha : M \to M_\alpha\).
Există \(g : M \to P\) astfel încât \(g_{\mid N} = f\).

\[
\begin{tikzcd}
0 \arrow[r] &
N \arrow[r,hook] \arrow[d,"f"'] &
M \arrow[dl,bend left=5,"g"'] \arrow[ddl,bend left=5,"g_\alpha"] \\
& P \arrow[d,"\pi_\alpha"'] & \\
& M_\alpha &
\end{tikzcd}
\]

\noindent\("\Rightarrow"\) Fie \(\forall \alpha \in \Lambda\) şi \(f : N \to M_\alpha\).
Considerând incluziunea canonică
\(\varepsilon_\alpha : M_\alpha \to P\),
cum \(P\) este \(M\)-injectiv, există \(g : M \to P\) care îl extinde pe
\(\varepsilon_\alpha \circ f : N \to P\).
Atunci \(\varepsilon_\alpha : M_\alpha \to P\) îl extinde pe \(f\) şi deci
\(M_\alpha\) este \(M\)-injectiv.


\[
\begin{tikzcd}
0 \arrow[r] &
N \arrow[r, hook] \arrow[d,"f"'] &
M \arrow[ddl,bend left=1,"g"'] \arrow[dddl,bend left=9,"\pi_\alpha g"] \\
& M_\alpha \arrow[d,"\varepsilon_\alpha"'] & \\
& P \arrow[d,"\pi_\alpha"'] & \\
& M_\alpha &
\end{tikzcd}
\]

\end{proof}

\begin{corollary}
\leavevmode
\begin{enumerate}
  \item Fie \((Q_\alpha)_{\alpha \in \Lambda}\) o familie de \(R\)-module.
        Atunci \(\prod_{\alpha \in \Lambda} Q_\alpha\) este injectiv
        dacă şi numai dacă \(Q_\alpha\) este injectiv pentru orice \(\alpha\in\Lambda\).
  \item \(Q_1 \oplus Q_2\) este \(R\)-modul injectiv dacă şi numai dacă
        \(Q_i\) este injectiv pentru \(i=1,2\).
        În particular, un sumand direct al unui modul injectiv este injectiv.
\end{enumerate}
\end{corollary}

\begin{proposition}
Fie \(Q\) un \(R\)-modul.
\begin{enumerate}
  \item Dacă \(0 \to M' \xrightarrow{f} M \xrightarrow{g} M'' \to 0\) este
        un şir exact de \(R\)-module şi \(Q\) este \(M\)-injectiv,
        atunci \(Q\) este \(M'\)-injectiv şi \(M''\)-injectiv.
  \item Dacă \((M_\alpha)_{\alpha \in \Lambda}\) este o familie de submodule ale lui \(M\)
        astfel încât \(M = \sum_{\alpha\in\Lambda} M_\alpha\) şi
        \(Q\) este \(M_\alpha\)-injectiv pentru orice \(\alpha\), atunci \(Q\) este \(M\)-injectiv.
  \item Fie \((N_\alpha)_{\alpha\in\Lambda}\) o familie de \(R\)-module.
        Atunci \(Q\) este \(\bigoplus_{\alpha\in\Lambda} N_\alpha\)-injectiv
        dacă şi numai dacă \(Q\) este \(N_\alpha\)-injectiv pentru orice \(\alpha\in\Lambda\).
\end{enumerate}
\end{proposition}


\begin{proof}
Pentru a arăta că \(Q\) este \(M'\)-injectiv, considerăm \(N\) un submodul al lui
\(M'\) și \(\varphi : N \rightarrow Q\) un morfism de \(R\)-module. Cum \(Q\) este
\(M\)-injectiv, există \(\phi : M \to Q\) astfel încât \(g \circ f\!\mid_{N} = \varphi\)
și deci \(\psi \circ f : M' \to Q\) este un morfism care îl extinde pe \(\varphi\).
\[
\begin{tikzcd}
0 \arrow[r] &
N \arrow[r] \arrow[d,"\varphi"'] &
M' \arrow[r,"f"] \arrow[dl] &
M \arrow[dll,bend left=5,"\psi"'] \\
& Q & &
\end{tikzcd}
\]

Fie \(h : L \to M''\) un monomorfism. Putem presupune, fără a restrânge generalitatea,
că \(M' \le M\) și \(M'' = M / M'\). Cum \(L \cong h(L) \le M''\), există
\(P \le M\), \(M' \subseteq P\) astfel încât \(h(L) = P / M'\) și deci \(L \cong P/M'\).
Obținem diagrama comutativă:

\[
\begin{tikzcd}
& & 0 \arrow[d] & 0 \arrow[d] & \\
0 \arrow[r] &
M' \arrow[r] \arrow[d, equal] &
P \arrow[r] \arrow[d] &
L \arrow[r] \arrow[d,"h"] &
0 \\
0 \arrow[r] &
M' \arrow[r] &
M \arrow[r] &
M'' \arrow[r] &
0
\end{tikzcd}
\]

Cum \(Q\) este \(M\)-injectiv, aplicând functorul \(\Hom(-,Q)\) obținem diagrama comutativă:


\[
\begin{tikzcd}
0 \arrow[r] &
\Hom(M'',Q) \arrow[r] \arrow[d,"h^{*}"'] &
\Hom(M,Q) \arrow[r] \arrow[d] &
\Hom(M',Q) \arrow[r] \arrow[d,equal] &
0 \\
0 \arrow[r] &
\Hom(L,Q) \arrow[r] &
\Hom(P,Q) \arrow[r] \arrow[d] &
\Hom(M',Q) &
\\
& & 0 & &
\end{tikzcd}
\]


Obținem că \(h^{*} = \Hom(h,Q)\) este epimorfism, ceea ce arată că
\(Q\) este \(M''\)-injectiv.


\medskip

2) Fie \(N\) un submodul al lui \(M\) şi \(f : N \to Q\) un morfism de \(R\)-module.
Considerăm mulţimea
\[
\mathfrak{S} = \{(L,h) \mid N \le L \le M,\; h : L \to Q,\; h\!\mid_{N} = f\}.
\]
Cum \((N,f) \in \mathfrak{S}\), avem \(\mathfrak{S} \ne \varnothing\).
Definim pe \(\mathfrak{S}\) relaţia de ordine
\((L_1,h_1) \preccurlyeq (L_2,h_2)\) dacă şi numai dacă
\(L_1 \le L_2\) şi \(h_2\!\mid_{L_1} = h_1\).
Se observă că \(\mathfrak{S}\) este inductivă şi, din lema lui Zorn,
rezultă că există \((L_0,g_0)\) element maximal al lui \(\mathfrak{S}\).
Pentru a arăta că \(L_0 = M\) este suficient să arătăm că
\(M_\alpha \le L_0\) pentru orice \(\alpha \in \Lambda\).

Considerând diagrama

\[
\begin{tikzcd}
0 \arrow[r] &
L_0 \cap M_\alpha \arrow[r,"i_\alpha"] \arrow[d,"i_0"'] &
M_\alpha \arrow[ddl,"h_\alpha"] \\
& L_0 \arrow[d,"g_0"'] & \\
& Q &
\end{tikzcd}
\]

rezultă că există \(h_\alpha : M_\alpha \to Q\) astfel încât
\(h_\alpha \circ i_\alpha = g_0 \circ i_\alpha\).
Definim
\(h^{*} : L_0 + M_\alpha \to Q\),
\(h^{*}(l + m_\alpha) = g_0(l) + h_\alpha(m_\alpha)\),
pentru orice \(l \in L_0\), \(m_\alpha \in M_\alpha\).
Dacă \(l + m_\alpha = 0\), atunci \(l = -m_\alpha \in L_0 \cap M_\alpha\) şi deci
\(h^{*}(l + m_\alpha) = g_0(l) + h_\alpha(l)\), ceea ce arată că \(h^{*}\) este bine definită.
Atunci \((L_0 + M_\alpha, h^{*}) \in \mathfrak{S}\) şi, cum
\((L_0,g_0) \preccurlyeq (L_0 + M_\alpha, h^{*})\),
din maximalitatea lui \((L_0,g_0)\) rezultă că
\(L_0 = L_0 + M_\alpha\), adică \(M_\alpha \le L_0\) pentru orice \(\alpha \in \Lambda\).

3) „\(\Rightarrow\)” Cum \(N_\alpha \le N\) şi \(Q\) este \(N\)-injectiv,
avem că \(Q\) este \(N_\alpha\)-injectiv pentru orice \(\alpha \in \Lambda\).

„\(\Leftarrow\)” Fie \(N'_\alpha = i_\alpha(N_\alpha)\).
Cum \(Q\) este \(N_\alpha\)-injectiv şi \(N'_\alpha \cong N_\alpha\),
rezultă că \(Q\) este \(N'_\alpha\)-injectiv.
Apoi aplicăm (2).
\end{proof}

\begin{theorem}[Criteriul lui Baer]\label{thm:Baer-ro}
Pentru un \(R\)-modul \(Q\) următoarele afirmaţii sunt echivalente:
\begin{enumerate}
  \item \(Q\) este injectiv.
  \item \(Q\) este \(R\)-injectiv.
  \item Pentru orice ideal drept \(I\) al lui \(R\) şi orice morfism \(f : I \to Q\)
        există \(x \in Q\) astfel încât \(f(a) = xa\) pentru orice \(a \in I\).
\end{enumerate}
\end{theorem}

\begin{proof}
Implicaţia \((1) \Rightarrow (2)\) este evidentă.

\((2) \Rightarrow (1)\).
Fie \(M\) un \(R\)-modul şi \(x \in M\).
Cum \(\varphi_x : R \to xR\), \(\varphi_x(a) = xa\) pentru orice \(a \in R\),
este morfism surjectiv de \(R\)-module, rezultă că
\(R / \Ker \varphi_x \cong xR\).
Cum \(Q\) este \(R\)-injectiv, din 2.1.4(1) rezultă că
\(Q\) este \(R / \Ker \varphi_x\)-injectiv şi deci
\(Q\) este \(xR\)-injectiv pentru orice \(x \in M\).
Atunci, din 2.1.6 obţinem că \(Q\) este \(M\)-injectiv.

\((2) \Rightarrow (3)\).
Fie \(I\) un ideal drept al lui \(R\) şi \(f : I \to Q\).
Există \(g : R \to Q\) astfel încât \(g\!\mid_{I} = f\).
Fie \(x = g(1) \in Q\).
Atunci \(f(a) = g(a) = ag(1) = xa\) pentru orice \(a \in I\).

\((3) \Rightarrow (2)\).
Dacă pentru un morfism \(f : I \to Q\) există \(x \in Q\) cu \(f(a) = xa\)
pentru orice \(a \in I\), atunci, definind
\(g : R \to Q\) prin \(g(r) = xr\) pentru orice \(r \in R\),
rezultă că \(g\!\mid_{I} = f\).
\end{proof}

\begin{definition}
\leavevmode
\begin{enumerate}
  \item Un \(R\)-modul \(Q\) se numeşte \emph{divizibil} dacă pentru orice
        \(y \in Q\) şi orice \(a \in R\) nendivizor al lui zero, există
        \(x \in Q\) astfel încât \(ax = y\).
        Se verifică uşor că orice modul factor al unui modul divizibil este divizibil.
   \item Un domeniu de integritate comutativ se numeşte \emph{PID-inel}
        dacă orice ideal al său este principal.
\end{enumerate}
\end{definition}


\begin{proposition}
\leavevmode
\begin{enumerate}
\item Orice modul injectiv este divizibil.
\item Fie \(R\) un PID-inel.
  \begin{enumerate}
    \item[(i)] Dacă \(Q\) este un \(R\)-modul, atunci \(Q\) este injectiv dacă şi numai dacă este divizibil.
    \item[(ii)] Dacă \(I\) este un ideal nenul al lui \(R\), atunci \(R/I\) este \(R\)-modul quasi-injectiv. 
          În particular, \(\mathbb{Z}_n\) este \(\mathbb{Z}\)-modul quasi-injectiv, \(\forall n \ge 1\).
  \end{enumerate}
\end{enumerate}
\end{proposition}


\begin{proof}
1) Fie \(Q\) un \(R\)-modul divizibil, \(y \in Q\) şi \(a \in R\) nendivizor al lui zero. Definim  
\(f : aR \to Q\) prin \(f(ax) = yx\) pentru orice \(x \in R\).  
Cum \(a\) este nendivizor al lui zero, \(f\) este bine definită.  
Folosind criteriul lui Baer rezultă că există \(x \in Q\) astfel încât  
\(f(\lambda) = x\lambda\), pentru orice \(\lambda \in I\).  
Deci \(y = f(a) = f(a \cdot 1) = xa\).

2) \((i)\) Implicaţia „\(\Rightarrow\)” este evidentă din (1).

„\(\Leftarrow\)” Fie \(Q\) un \(R\)-modul divizibil şi \(I\) un ideal drept al lui \(R\).  
Atunci există \(a \in R\) astfel încât \(I = aR\).  
Considerăm \(f : I \to Q\) un morfism de \(R\)-module.  
Există \(x \in Q\) astfel încât \(f(a) = xa\).  
Atunci \(f(ar) = xar\) pentru orice \(r \in R\) şi deci, conform criteriului lui Baer,  
\(Q\) este injectiv.

(ii) Fie \(I = aR\) şi \(J = bR\) ideale nenule ale lui \(R\) astfel încât \(I \subseteq J\) şi  
\(f : J/I \to R/I\) un morfism de \(R\)-module.  
Există \(c \in R\) astfel încât \(a = bc\).  
Dacă \(f(\bar{b}) = \hat{x} \in R/I\), atunci \(x \in I\) şi deci există \(a_1 \in R\) astfel încât  
\(x = aa_1\).  
Rezultă că \(x = ba_1\).  
Definim \(g : R/I \to R/I\), \(g(\bar{r}) = \widehat{ar_1}\) pentru orice \(r \in R\).  
Atunci \(g\) este morfism de \(R\)-module şi \(g(\bar{b}) = \hat{x}\),  
deci \(g\!\mid_{J/I} = f\), ceea ce arată că \(R/I\) este quasi-injectiv.
\end{proof}

\begin{lemma}
Fie inelele \(A, S, T\) şi bimodulele \({}_S M_A\), \({}_A N_T\).  
Atunci \(\Hom_A(M,N)\) are o structură de bimodul \(S\)-stâng şi \(T\)-drept prin operaţiile:
\[
(s \cdot f)(x) = f(xs), \qquad (f \cdot t)(x) = f(x)t,
\]
unde \(s \in S\), \(t \in T\), \(x \in M\), \(f \in \Hom_A(M,N)\).
\end{lemma}


\begin{proof}
    
Fie \(a,b \in A\) şi \(x,y \in M\). Atunci:
\[
(s \cdot f)(ax + by)
= f((ax+by)s)
= f(a(xs)) + f(b(ys)) = a f(xs) + b f(ys)
= a (s \cdot f)(x) + b (s \cdot f)(y),
\]
şi deci \(s \cdot f \in \Hom_A(M,N)\).  
Analog \(f \cdot t \in \Hom_A(M,N)\).

Pentru \(s,s' \in S\) şi \(f,g \in \Hom_A(M,N)\):
\[
(s \cdot (f+g))(x) = (f+g)(xs) = f(xs) + g(xs) = (s\cdot f)(x) + (s \cdot g)(x).
\tag{1}
\]

\[
((s+s') \cdot f)(x) = f(x(s+s')) = f(xs + xs') = f(xs) + f(xs') = (s\cdot f)(x) + (s' \cdot f)(x).
\tag{2}
\]

\[
((ss') \cdot f)(x) = f(x(ss')) = f((xs)s') = (s' \cdot f)(xs) = (s \cdot (s' \cdot f))(x).
\tag{3}
\]

\[
(1_S \cdot f)(x) = f(x1_S) = f(x).
\tag{4}
\]

Din (1)–(4) avem că \(\Hom_A(M,N)\) este \(S\)-modul stâng.  
La fel se arată că este şi \(T\)-modul drept.  
Prin urmare \(\Hom_A(M,N)\) este un bimodul \(S\)-stâng şi \(T\)-drept.
\end{proof}


\begin{proposition}[Eckmann–Schopf]
Fie \(Q\) un grup abelian divizibil. Atunci \(R\)-modulul stâng \(\Hom_\mathbb{Z}(R,Q)\) este injectiv.
\end{proposition}

\begin{proof}
Conform lemei precedente, $\Hom_{\mathbb{Z}}(R,Q)$ are o structură de
$R$-modul stâng dată de operaţia
\[
(r \cdot f)(a) = f(ar), \qquad \forall a,r \in R,\ f \in \Hom_{\mathbb{Z}}(R,Q).
\]
Fie $I$ un ideal stâng al lui $R$ şi $h : I \to \Hom_{\mathbb{Z}}(R,Q)$
un morfism de $R$-module stângi. Atunci aplicaţia
\[
\gamma : \mathbb{Z}I \longrightarrow \mathbb{Z}Q, \qquad
\gamma(a) = h(a)(1)
\]
defineşte un morfism de $\mathbb{Z}$–module.
Cum $Q$ este $\mathbb{Z}$–injectiv, există
$\widetilde{\gamma} : \mathbb{Z}R \to \mathbb{Z}Q$ astfel încât
$\widetilde{\gamma}_{\mid I} = \gamma$.
Pentru $a \in I$ şi $r \in R$ avem
\[
(a \cdot \widetilde{\gamma})(r)
  = \widetilde{\gamma}(ra)
  = h(ra)(1)
  = (r \cdot h(a))(1)
  = h(a)(r),
\]
deci $h(a) = a \cdot \widetilde{\gamma}$ pentru orice $a \in I$.
Conform criteriului lui Baer, $\Hom_{\mathbb{Z}}(R,Q)$ este
$R$–modul stâng injectiv.
\end{proof}


\begin{proposition}
Orice $R$-modul stâng $M$ poate fi scufundat într-un
$R$-modul stâng injectiv.
\end{proposition}

\begin{proof}
Există un $\mathbb{Z}$-modul liber de forma $\mathbb{Z}^{(A)}$ şi
un $\mathbb{Z}$-morfism surjectiv $f : \mathbb{Z}^{(A)} \to M$.
Atunci
\[
\mathbb{Z}M \cong \mathbb{Z}^{(A)} / \ker f
  \subseteq \mathbb{Q}^{(A)} / \ker f,
\]
şi deci există un grup abelian divizibil $G$ astfel încât
$\mathbb{Z}M \subseteq \mathbb{Z}G$.
Aplicând functorul $\Hom_{\mathbb{Z}}(R,-)$ obţinem un monomorfism
\[
{}_R M \cong \Hom_{\mathbb{Z}}(R,M)
   \hookrightarrow \Hom_{\mathbb{Z}}(R,G).
\]
Cum $G$ este divizibil, din Propoziţia~2.1.13 rezultă că
$\Hom_{\mathbb{Z}}(R,G)$ este $R$-modul stâng injectiv.
Prin urmare $M$ se scufundă într-un $R$-modul stâng injectiv.
\end{proof}


\begin{proposition}
Fie $Q$ un $R$-modul. Atunci $Q$ este injectiv dacă şi numai dacă
orice şir exact de forma
\[
0 \longrightarrow Q \xrightarrow{f} M \xrightarrow{g} M' \longrightarrow 0
\]
este scindat.
\end{proposition}

\begin{proof}
„$\Rightarrow$”\;
Presupunem că $Q$ este injectiv.
Din exactitatea şirului avem că $f$ este monomorfism.
Prin injectivitatea lui $Q$ există $h : M \to Q$ astfel încât
$h f = \mathrm{id}_Q$, ceea ce arată că şirul este scindat.

„$\Leftarrow$”\;
Folosind Propoziţia~2.1.14, există un $R$-modul injectiv $Q'$ şi
un monomorfism $i : Q \to Q'$.
Avem un şir exact
\[
0 \longrightarrow Q \xrightarrow{i} Q' \longrightarrow Q'/i(Q) \longrightarrow 0.
\]
Prin ipoteză acest şir este scindat, deci $Q$ este sumand direct în $Q'$.
Un sumand direct al unui modul injectiv este injectiv, aşadar $Q$ este injectiv.
\end{proof}



\section{Anvelope injective}

\begin{definition}
Fie $M$ un $R$-modul.
O pereche $(E,i)$ se numeşte \emph{anvelopă injectivă} a lui $M$ dacă
$E$ este modul injectiv şi $i : M \to E$ este un monomorfism esenţial.
\end{definition}

\begin{proposition}
Fie $Q$ un $R$-modul injectiv.
Atunci orice submodul complement al lui $Q$ este sumand direct în $Q$.
\end{proposition}

\begin{proof}
Fie $K$ un submodul al lui $Q$ şi $N$ un complement al lui $K$ în $Q$,
adică $K \cap N = 0$ şi $K+N$ este submodul esenţial în $Q$.
Atunci $(K+N)/N \cong Q/N$.
Definim $g : (K+N)/N \to Q$ prin
\[
g((x+y)+N) = x, \qquad x \in K,\ y \in N.
\]
Cum $K \cap N = 0$, aplicaţia $g$ este bine definită şi este monomorfism.
Injectivitatea lui $Q$ asigură existenţa unui morfism
$h : Q/N \to Q$ astfel încât $h_{\mid (K+N)/N} = g$.
Deoarece $(K+N)/N \cong Q/N$ şi $g$ este monomorfism, şi $h$ este monomorfism.
Avem $K = \operatorname{Im} g = h((K+N)/N) \subseteq h(Q/N)$.
Cum $K$ este submodul închis, rezultă $K = h(Q/N)$.
Din $h$ monomorfism obţinem $(K+N)/N = Q/N$, deci $K+N = Q$.
Prin urmare $K$ este un sumand direct în $Q$.
\end{proof}

\begin{theorem}[Eckmann--Schopf]
Orice $R$-modul $M$ are o anvelopă injectivă unică până la un izomorfism.
\end{theorem}

\begin{proof}
Din Propoziţia~2.1.14 există un $R$-modul injectiv $Q$ astfel încât
$M \leq Q$.
Fie $E$ o extensie esenţială maximală a lui $M$ în $Q$.
Atunci $E$ este un submodul complement în $Q$, iar din propoziţia
precedentă rezultă că $E$ este injectiv.
Astfel $(E,i)$, cu $i : M \hookrightarrow E$ incluziunea, este o anvelopă
injectivă a lui $M$.

Pentru unicitate, fie $(E_1,i_1)$ şi $(E_2,i_2)$ două anvelope injective
ale lui $M$.
Cum $E_2$ este injectiv, există $f : E_1 \to E_2$ astfel încât
$f i_1 = i_2$.
Morfismul $i_2$ este monomorfism, iar $i_1$ este monomorfism esenţial,
deci (folosind 1.4) rezultă că $f$ este monomorfism.
Avem $E_1 \cong f(E_1)$ şi $E_2 = f(E_1) \oplus E_3$ pentru un anumit
submodul $E_3$.
Dar $i_2(M) \subseteq f(E_1)$, deci $i_2(M) \cap E_3 = 0$.
Cum $i_2$ este monomorfism esenţial, rezultă $E_3 = 0$, deci
$E_2 = f(E_1)$ şi $f$ este izomorfism.
\end{proof}

În practică vom considera un reprezentant al acestei clase pe care îl
vom nota $E(M)$ astfel încât $M \ess E(M)$.

\begin{proposition}
Fie $M$ un $R$-modul şi $i : M \to Q$ un monomorfism cu $Q_R$ injectiv.
Următoarele afirmaţii sunt echivalente:
\begin{enumerate}
  \item $(Q,i)$ este o anvelopă injectivă a lui $M$;
  \item pentru orice monomorfism $f : M \to Q'$ cu $Q'$ injectiv,
        există un monomorfism $g : Q \to Q'$ astfel încât $g i = f$.
\end{enumerate}
\end{proposition}

\begin{proof}
$(1) \Rightarrow (2)$.
Fie $f : M \to Q'$ un monomorfism cu $Q'$ injectiv.
Prin injectivitatea lui $Q'$ există $u : Q \to Q'$ astfel încât
$u i = f$.
Cum $i$ este monomorfism esenţial şi $Q'$ este injectiv, imaginea
$u(Q)$ este un complement al lui $f(M)$, iar din definiţia anvelopei
injective rezultă că $u$ este monomorfism; punem $g=u$.

$(2) \Rightarrow (1)$.
Fie $(E(M),j)$ o anvelopă injectivă a lui $M$.
Aplicând (2) la $f=j$ obţinem un monomorfism
$g : Q \to E(M)$ cu $g i = j$.
Cum $j$ este monomorfism esenţial, rezultă că şi $i$ este monomorfism
esenţial, deci $(Q,i)$ este o anvelopă injectivă a lui $M$.
\end{proof}

\begin{proposition}
Oricare ar fi $R$-modulele drepte $M_1,M_2,\dots,M_n$ avem
\[
E\!\left(\bigoplus_{i=1}^{n} M_i\right)
  \cong \bigoplus_{i=1}^{n} E(M_i).
\]
\end{proposition}

\begin{proof}
Din 1.9, $\bigoplus_{i=1}^{n} E(M_i)$ este o extensie esenţială
a lui $\bigoplus_{i=1}^{n} M_i$.
Cum
\[
\bigoplus_{i=1}^{n} E(M_i) \cong \prod_{i=1}^{n} E(M_i),
\]
din 2.1.3 rezultă că $\bigoplus_{i=1}^{n} E(M_i)$ este injectiv.
Prin unicitatea anvelopei injective obţinem
\[
E\!\left(\bigoplus_{i=1}^{n} M_i\right)
  \cong \bigoplus_{i=1}^{n} E(M_i).
\]
\end{proof}

\begin{theorem}
Fie \(Q\) şi \(M\) două \(R\)-module. Atunci \(Q\) este \(M\)-injectiv dacă şi numai dacă
\(f(M) \leq Q\), oricare ar fi \(f \in \Hom(E(M),E(Q))\).
\end{theorem}

\begin{proof}
„\(\Rightarrow\)” Fie \(f \in \Hom(E(M),E(Q))\) şi notăm
\[
K := \{\,m \in M \mid f(m) \in Q\,\}.
\]
Cum \(Q\) este \(M\)-injectiv, există un morfism \(\bar{f} : M \to Q\) astfel încât
\(\bar{f}\!\mid_{K} = f\!\mid_{K}\).
Arătăm că
\[
Q \cap (\bar{f}-f)(M) = 0.
\]
Fie \(x \in Q\) şi \(m \in M\) cu \(x = (\bar{f}-f)(m)\).
Atunci
\[
f(m) = \bar{f}(m) - x \in Q,
\]
deci \(m \in K\).
Urmează că
\[
x = \bar{f}(m) - f(m) = f(m) - f(m) = 0.
\]
Prin urmare \(Q \cap (\bar{f}-f)(M) = 0\) şi, cum \(Q \ess E(Q)\), rezultă că
\((\bar{f}-f)(M) = 0\).
Aşadar \(f(M) = \bar{f}(M) \leq Q\).

„\(\Leftarrow\)” Întrucât \(E(Q)\) este injectiv, este suficient să considerăm
\(f \in \Hom(M,E(Q))\).
Fie \(N\) un submodul al lui \(M\) şi \(g : N \to Q\) un morfism de \(R\)-module.
Cum \(E(Q)\) este injectiv, există un morfism \(\tilde{g} : M \to E(Q)\) astfel încât
\(\tilde{g}\!\mid_{N} = i \circ g\), unde \(i : Q \to E(Q)\) este injecţia canonică.
Prin ipoteză avem \(\tilde{g}(M) \leq Q\), astfel încât, identificând \(\tilde{g}\) cu
corestricţia sa la \(Q\), obţinem un morfism \(h : M \to Q\) cu
\(h\!\mid_{N} = g\).
Prin urmare \(Q\) este \(M\)-injectiv.
\end{proof}

\begin{theorem}[Matlis--Bass]
Fie \(R\) un inel. Atunci \(R\) este noetherian la dreapta dacă şi numai dacă,
pentru orice \(R\)-modul simplu \(S_i\) (\(i \geq 1\)),
\[
Q := \bigoplus_{i=1}^{\infty} E(S_i)
\]
este un \(R\)-modul injectiv.
\end{theorem}

\begin{proof}
„\(\Rightarrow\)” Fie \(L\) un ideal drept al lui \(R\),
\[
Q = \bigoplus_{i=1}^{\infty} E(S_i)
\]
şi \(f : L \to Q\) un morfism de \(R\)-module.
Există elemente \(a_1,\dots,a_n \in L\) astfel încât
\[
L = a_1 R + a_2 R + \cdots + a_n R.
\]
În mod evident există \(m \geq 1\) astfel încât
\(f(a_k) \in \bigoplus_{j=1}^{m} E(S_j)\) pentru orice \(k = 1,\dots,n\),
deci
\[
\Im f \subseteq \bigoplus_{j=1}^{m} E(S_j).
\]
Cum \(\bigoplus_{j=1}^{m} E(S_j)\) este injectiv, există
\(g : R \to \bigoplus_{j=1}^{m} E(S_j)\) astfel încât \(g\!\mid_{L} = f\).
Notăm \(\bar{f} = i \circ g\), unde
\(i : \bigoplus_{j=1}^{m} E(S_j) \to Q\) este injecţia canonică.
Atunci \(\bar{f}\!\mid_{L} = f\), deci \(Q\) este injectiv.

„\(\Leftarrow\)” Presupunem că \(R\) nu este noetherian la dreapta.
Atunci există un şir strict ascendent de ideale la dreapta, finit generate:
\[
L_1 \subsetneq L_2 \subsetneq \cdots \subsetneq L_n \subsetneq \cdots .
\]
Din lema lui Krull rezultă că, pentru orice \(n \geq 1\), există un submodul
maximal \(M_n \subsetneq L_n\) astfel încât
\[
L_{n-1} \subseteq M_n \quad \text{pentru orice } n \geq 2.
\]
Fie
\[
L := \bigcup_{k=1}^{\infty} L_k, \qquad
\pi_k : L_k \longrightarrow L_k/M_k
\]
proiecţiile canonice şi
\[
E_k := E(L_k/M_k) \quad \text{pentru orice } k \geq 1.
\]
Atunci
\[
E := \bigoplus_{k=1}^{\infty} E_k
\]
este injectiv şi
\[
f : L \longrightarrow E, \qquad
f(a) = \sum_{k=1}^{\infty} \pi_k(a)
\]
este bine definit.
Există un element
\(x \in E_1 \oplus \cdots \oplus E_n\) astfel încât
\(f(a) = x a\) pentru orice \(a \in L\).
Rezultă că \(\pi_k(a) = 0\) pentru orice \(k \geq n+1\),
adică \(a \in M_k\) pentru orice \(k \geq n+1\).
Prin urmare
\[
L \subseteq M_{n+1} \subsetneq L_{n+1} \subseteq M_{n+2} \subsetneq L_{n+2}
\subseteq \cdots \subseteq L,
\]
contradicţie.
Obţinem că \(R\) este noetherian la dreapta.
\end{proof}



\chapter{Sume directe de module coireductibile}

\begin{proposition}
Fie \(M\) un \(R\)-modul şi \(E(M)\) anvelopa sa injectivă. Atunci următoarele
afirmaţii sunt echivalente:
\begin{enumerate}
  \item \(E(M)\) este indecompozabil.
  \item Dacă \(L\) şi \(K\) sunt submodule nenule ale lui \(M\), atunci
        \(L \cap K \neq 0\).
  \item Dacă \(x,y \in M\setminus\{0\}\), atunci există \(a,b \in R\) astfel
        încât \(0 \neq xa = yb\).
  \item \(M\) este o extensie esenţială a oricărui submodul nenul al său.
\end{enumerate}
\end{proposition}

\begin{proof}
\((2) \Rightarrow (1)\).
Presupunem că \(E(M) = L' \oplus K'\) cu \(L'\) şi \(K'\) submodule nenule ale
lui \(E(M)\).
Cum \(M \ess E(M)\), avem că \(L = L' \cap M \neq 0\) şi
\(K = K' \cap M \neq 0\), deci
\[
L \cap K = (L' \cap K') \cap M = 0,
\]
ceea ce contrazice ipoteza (2).

\smallskip
\((1) \Rightarrow (2)\).
Presupunem că există \(L,K\) submodule nenule ale lui \(M\) astfel încât
\(L \cap K = 0\).
Atunci \(E(L) \leq E(M)\) şi \(K \cap E(L) = 0\) (altfel, cum
\(L \subseteq E(L)\), am avea \(L \cap K \neq 0\)).
Dar şirul exact scurt
\[
0 \longrightarrow E(L) \longrightarrow E(M) \longrightarrow E(M)/E(L)
\longrightarrow 0
\]
este scindat deoarece \(E(L)\) este injectiv şi deci
\(E(M) \cong E(L) \oplus E(M)/E(L)\).
Obţinem astfel
\[
0 = K \cap E(L) = K \cap E(M) = K,
\]
contradicţie.

Echivalenţele \((2) \Leftrightarrow (3)\) şi \((2) \Leftrightarrow (4)\) sunt
evidente.
\end{proof}

\begin{definition}
Un modul \(M\) care satisface una din condiţiile echivalente de mai sus se
numeşte \emph{coireductibil} (sau \emph{uniform}).
Fie \(M\) un \(R\)-modul şi \(N\) un submodul propriu al său.
Dacă \(M/N\) este coireductibil, atunci spunem că \(N\) este
\emph{ireductibil} în \(M\).
Este clar că \(N\) este ireductibil în \(M\) dacă şi numai dacă din egalitatea
\(N = P \cap Q\), unde \(P,Q \le M\), rezultă \(N = P\) sau \(N = Q\).
\end{definition}

\begin{example*}
\leavevmode
\begin{enumerate}
  \item Este clar că orice modul simplu este coireductibil.
  \item \(\mathbb{Z}\) este \(\mathbb{Z}\)-modul coireductibil.
  \item Din Propoziţia~3.1 rezultă că un modul injectiv este indecompozabil
        dacă şi numai dacă este coireductibil.
        În particular, dacă \(M\) este un \(R\)-modul coireductibil, atunci
        \(E(M)\) este coireductibil. Rezultă că \(\mathbb{Q}\) şi
        \(\mathbb{Z}_{p^\infty}\) sunt \(\mathbb{Z}\)-module coireductibile.
  \item \(\mathbb{Z}_n\) este \(\mathbb{Z}\)-modul coireductibil dacă şi numai
        dacă \(n\) este puterea unui număr prim. Dacă \(n = p^k\) cu
        \(p \ge 2\) şi \(k \ge 1\), atunci
        \(\langle p^m \rangle \subseteq \langle p^i \rangle \cap \langle p^j \rangle\)
        pentru orice \(i,j \in \{1,\dots,k-1\}\), unde \(m = \min(i,j)\), şi deci
        \(\mathbb{Z}_{p^k}\) este coireductibil.
        Reciproc, presupunem că \(\mathbb{Z}_n\) este coireductibil şi
        \(n = p_1^{\alpha_1}\cdots p_s^{\alpha_s}\) este descompunerea lui \(n\)
        în factori primi. Dacă \(s \ge 2\), atunci avem
        \[
          \langle p_1^{\alpha_1}\cdots p_{s-1}^{\alpha_{s-1}} \rangle
          + \mathbb{Z}
          \quad\text{şi}\quad
          \langle p_s^{\alpha_s} \rangle + \mathbb{Z}
        \]
        submodule nenule a căror intersecţie este zero, contradicţie.
        Obţinem că \(s = 1\), deci \(n\) este o putere a unui număr prim.
\end{enumerate}
\end{example*}

\begin{proposition}
Fie \(M\) un \(R\)-modul şi \(N\) un submodul propriu al lui \(M\).
Dacă \(x \in M \setminus N\), atunci există un submodul ireductibil \(P\) al
lui \(M\) astfel încât \(N \subseteq P\) şi \(x \notin P\).
\end{proposition}

\begin{proof}
Fie
\[
\mathfrak{S} = \{\,N' \le M \mid N \subseteq N' \text{ şi } x \notin N'\,\}.
\]
Mulţimea \(\mathfrak{S}\), ordonată prin incluziune, este inductivă, deci,
prin lema lui Zorn, admite un element maximal \(P\).
Arătăm că \(P\) este ireductibil în \(M\).
Presupunem că \(P = U \cap V\) cu \(U,V \le M\) şi \(P \subsetneq U\),
\(P \subsetneq V\).
Cum \(x \notin P\), avem \(x \notin U\) sau \(x \notin V\); să spunem
\(x \notin U\).
Atunci \(U \in \mathfrak{S}\) şi \(P \subsetneq U\), ceea ce contrazice
maximalitatea lui \(P\).
Rezultă că \(P\) este ireductibil.
\end{proof}

\begin{corollary}
Fie \(M\) un \(R\)-modul şi \(N\) un submodul propriu al lui \(M\).
Atunci \(N\) este intersecţie de submodule ireductibile în \(M\).
\end{corollary}

\begin{proof}
Rezultă imediat din propoziţia anterioară.
\end{proof}


\begin{corollary}
Let \(M\) be an \(R\)-module and \(N\) a proper submodule of \(M\).
Then \(N\) is the intersection of irreducible submodules of \(M\).
\end{corollary}

\begin{proof}
This follows immediately from Proposition~3.3, applied to each
\(x \in M \setminus N\).
\end{proof}

\begin{proof}
1) Rezultă imediat din 3.1.

2) Fie $M \ess E$ o extensie esenţială a lui $M$. Dacă $E_1,E_2$ sunt
submodule nenule ale lui $E$, atunci $M \cap E_1 \neq 0$ şi
$M \cap E_2 \neq 0$, de unde rezultă că
\[
(M \cap E_1) \cap (M \cap E_2) \neq 0
\]
şi deci $E_1 \cap E_2 \neq 0$, adică $E$ este coireductibil.
\end{proof}


\begin{lemma}
Fie $M$ un $R$-modul şi $(M_\alpha)_{\alpha \in \Lambda}$ o familie
de submodule independente. Dacă $N$ este un submodul al lui $M$ astfel
încât
\[
N \cap \Bigl(\sum_{\alpha \in \Lambda} M_\alpha\Bigr) \neq 0,
\]
există un submodul nenul într-un $M_\alpha$ izomorf cu un submodul din $N$.
\end{lemma}

\begin{proof}
Dacă $\operatorname{Card}\Lambda = 1$, afirmaţia este evidentă.

Dacă $\operatorname{Card}\Lambda = 2$, punem $P = N \cap (M_1 + M_2)$.
Dacă $N \cap M_1 = 0$, atunci $P \cong (P + M_1)/M_1$ şi
\[
(P + M_1)/M_1 \leq (M_1 + M_2)/M_1 \cong M_2.
\]
Deci $P$ este izomorf cu un submodul din $M_2$.
Dacă $\Lambda$ este mulţime finită, procedând prin inducţie,
demonstraţia se reduce la cazul precedent.

Dacă $\Lambda$ este infinită, considerăm
$x \in N \cap \bigl(\sum_{\alpha \in \Lambda} M_\alpha\bigr)$,
$x \neq 0$. Atunci există $\alpha_1,\ldots,\alpha_n \in \Lambda$ astfel
încât
\[
xR \cap (M_{\alpha_1} + M_{\alpha_2} + \cdots + M_{\alpha_n}) \neq 0
\]
şi ne reducem la cazul în care $\Lambda$ este finită.
\end{proof}

\begin{remark}
Fie $M$ un $R$-modul drept şi $\Omega$ mulţimea de submodule
coireductibile din $M$. Este posibil ca $\Omega = \emptyset$.
Fie
\[
\mathcal{S} = \bigl\{\Omega' \subseteq \Omega \,\bigm|\, \sum_{N \in \Omega'}
N \text{ este directă} \bigr\}.
\]
Perechea $(\mathcal{S},\subseteq)$ este inductivă şi, din lema lui Zorn,
$\mathcal{S}$ are un element maximal $\Omega_0$.
Punem $S = \bigoplus_{N \in \Omega_0} N$.
Spunem că $S$ este o sumă directă maximală de submodule
coireductibile din $M$.
\end{remark}

\begin{proposition}
Fie $M$ un $R$-modul şi $S$ o sumă directă maximală de submodule
coireductibile din $M$. Dacă $N$ este un submodul nenul al lui $M$,
atunci următoarele afirmaţii sunt echivalente:
\begin{enumerate}
  \item $S \cap N \neq 0$;
  \item $N$ conţine un submodul coireductibil.
\end{enumerate}
\end{proposition}

\begin{proof}
Implicaţia \((1)\Rightarrow(2)\) rezultă din lema~3.6.

Implicaţia \((2)\Rightarrow(1)\) rezultă din maximalitatea lui $S$.
\end{proof}

\begin{theorem}
Fie $M$ un $R$-modul şi $S$ o sumă directă maximală de submodule
coireductibile din $M$. Există un submodul $K$ maximal printre
submodulele lui $M$ care nu conţin nici un coireductibil, cu
proprietăţile:
\begin{enumerate}
  \item $S + K$ este directă;
  \item $(S \oplus K) \ess M$.
\end{enumerate}
\end{theorem}

\begin{proof}
\begin{enumerate}
\item Fie
\[
\mathcal{S} = \{\,L \leq M \mid \forall L' \leq L,\; L' \text{ nu este
coireductibil}\,\}.
\]
Avem $\mathcal{S} \neq \emptyset$ deoarece $0 \in \mathcal{S}$ şi
$\mathcal{S}$ este inductivă. Din lema lui Zorn, $\mathcal{S}$ are un
element maximal $K$. Din Propoziţia~3.8 rezultă că $S \cap K = 0$.

\item Presupunem că $N \cap (S \oplus K) = 0$, unde $N$ este un
submodul nenul al lui $M$. Atunci $(N + K) \cap S = 0$ şi din
Propoziţia~3.8 rezultă că $N + K$ nu conţine nici un submodul
coireductibil, ceea ce contrazice maximalitatea lui $K$.
Obţinem că $(S \oplus K) \ess M$.
\end{enumerate}
\end{proof}

\begin{definition}
Fie $M$ un $R$-modul.
O intersecţie finită de submodule
\[
\bigcap_{i \in I} N_i \qquad (I \text{ finit})
\]
se numeşte \emph{redusă} dacă, oricare ar fi $i \in I$, avem
\[
\bigcap_{i \in I} N_i \neq
\bigcap_{\substack{j \in I \\ j \neq i}} N_j .
\]

Este clar că, dacă avem o intersecţie finită
$N = \bigcap_{i \in I} N_i$ de submodule ireductibile, ea poate fi
adusă întotdeauna la o intersecţie redusă de submodule ireductibile.
\end{definition}


% Teorema 3.11 (Kuros–Ore) – RO
\begin{theorem}[Teorema (Kuros–Ore)]
Fie \(M\) un \(R\)-modul şi \(N\) un submodul al lui \(M\). Presupunem că avem
intersecţiile finite reduse pentru \(N\):
\[
N = N_1 \cap N_2 \cap \dots \cap N_m = L_1 \cap L_2 \cap \dots \cap L_n ,
\]
unde \(N_i\) şi \(L_j\) sunt submodule ireductibile din \(M\). Atunci \(m = n\).
\end{theorem}

\begin{proof}
Notăm \(N_i' = N_1 \cap \dots \cap N_{i-1} \cap N_{i+1} \cap \dots \cap N_m\).
Avem că \(N \subseteq N_i'\) şi \(N = N_i \cap N_i'\). Fie
\(P_j = N_i \cap L_j\), \(\forall j = 1,\dots,n\).
Atunci \(N \subseteq P_j \subseteq N_i'\) şi \(P_j \subseteq L_j\),
de unde rezultă că
\[
N \subseteq \bigcap_{j=1}^n P_j \subseteq \bigcap_{j=1}^n L_j = N,
\]
şi deci \(\bigcap_{j=1}^n P_j = N\).

Cum \(N_i\) este ireductibil în \(M\), rezultă că \(N_i\) este ireductibil în
\(N_i + N_i'\), şi deci
\[
\frac{N_i'}{N_i' \cap N_i} \cong \frac{N_i + N_i'}{N_i}
\]
este coirreductibil, ceea ce arată că \(N\) este ireductibil în \(N_i'\).

Atunci, din egalitatea \(N = \bigcap_{j=1}^n P_j\),
există un \(j\) astfel încât \(N = P_j\).
Deci \(N = N_i \cap L_j\), adică
\[
N = N_i \cap \dots \cap N_{i-1} \cap L_j \cap N_{i+1} \cap \dots \cap N_m.
\]

Notăm \(I = \{1,2,\dots,m\}\),
\(J = \{1,2,\dots,n\}\) şi fie \(i_1 \in I\).
Există \(j_1 \in J\) astfel încât
\[
N = L_{j_1} \cap \left(\bigcap_{i \ne i_1} N_i\right).
\]

Deoarece intersecţia este finită, putem să o reducem:
din reducerea succesivă şi prin faptul că nu dispare niciun \(L_{j_k}\),
obţinem un şir strict descrescător
\[
J \supseteq J_1 \supseteq J_2 \supseteq \dots \supseteq J_r,
\]
cu cel mult \(r\le m\) elemente,
astfel încât pentru un anumit set final \(J'\subseteq J\),
\[
N = \bigcap_{j\in J'} L_j.
\]

Este clar că \(r\le m\). Cum intersecţia \(N = L_1 \cap \dots \cap L_n\)
este redusă, avem \(J' = J\).
Pe de altă parte, \(J'\) are cel mult \(r\) elemente,
deci \(n \le r \le m\), iar simetric \(m \le n\).
\end{proof}

% Teorema 3.12 – RO
\begin{theorem}
Fie \(M\) un \(R\)-modul şi \((N_i)_{i=1}^m\) şi \((L_j)_{j=1}^n\) două familii
independente de submodule coirreductibile astfel încât
\[
\bigoplus_{i=1}^m N_i \subseteq M
\quad\text{şi}\quad
\bigoplus_{j=1}^n L_j \subseteq M
\]
sunt esenţiale în \(M\).  
Atunci \(m = n\).
\end{theorem}

\begin{proof}
Notăm \(N_k' = \bigoplus_{i\ne k} N_i\), pentru fiecare \(k=1,\dots,m\).
Atunci \(N_k \cap N_k' = 0\).

Se verifică uşor că mulţimea
\[
\mathfrak{S}_k = \{P \le M \mid N_k \cap P = 0 \text{ şi } N_k \subseteq P\}
\]
este inductivă. Din lema lui Zorn, are un element maximal \(P_k\).
Dacă \(Q,Q' \le M\) sunt astfel încât \(P_k \subseteq Q \subseteq Q'\),
atunci \(N_k \cap Q \ne 0\) şi \(N_k \cap Q' \ne 0\),
iar ireductibilitatea lui \(N_k\) arată că \(P_k = Q = Q'\).
Deci \(P_k\) este coirreductibil pentru fiecare \(k\).

Fie \(k\in\{1,\dots,m\}\).
Atunci
\[
N = \bigoplus_{k=1}^m N_k = N_k \oplus N_k'.
\]
Aplicând legea distributivităţii modulare, obţinem
\[
N \cap P_k = (N_k \oplus N_k') \cap P_k
           = N_k + (N_k' \cap P_k),
\]
deci \(N \cap P_k = N_k\).
Rezultă că
\[
N \cap \bigcap_{k=1}^m P_k = \bigcap_{k=1}^m (N\cap P_k) = \bigcap_{k=1}^m N_k = 0.
\]

Dar \(NVM\), deci \(\bigcap_{k=1}^m P_k = 0\).

Cum \(N_i = \bigcap_{k\ne i} P_k\), avem că fiecare intersecţie este redusă.

Analog, pentru familia \((L_j)\), obţinem submodule ireductibile \((Q_j)\)
astfel încât \(\bigcap_{j=1}^n Q_j = 0\), reducând intersecţia.

Din teorema Kuros–Ore, rezultă \(m = n\).
\end{proof}


\begin{definition}
Spunem că un \(R\)-modul \(M\) are \emph{dimensiune coireductibilă finită} dacă
există o familie finită independentă \((N_i)_{i=1}^n\) de submodule
coireductibile din \(M\) astfel încât
\[
\bigoplus_{i=1}^n N_i \ess M.
\]
În acest caz numărul \(n\) se numeşte \emph{dimensiunea coireductibilă} a lui
\(M\) şi scriem \(\dim M = n\).
\end{definition}

\begin{theorem}
Fie \(M\) un \(R\)-modul. Următoarele afirmaţii sunt echivalente:
\begin{enumerate}
  \item \(M\) are dimensiune coireductibilă finită.
  \item \(M\) satisface condiţia lanţurilor ascendente pentru sume directe.
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{\((2) \Rightarrow (1)\).}
Fie \(\mathcal{S}'\) şi \(\mathcal{S}\) două familii finite de submodule
independente din \(M\).
Spunem că \(\mathcal{S}'\) este o \emph{rafinare} a lui \(\mathcal{S}\) dacă
oricare submodul din \(\mathcal{S}'\) este conţinut într-un submodul din
\(\mathcal{S}\).
Se verifică uşor că relaţia de rafinare este o relaţie de ordine pe
mulţimea familiilor de submodule ale lui \(M\).
Cum \(M\) satisface (2), există o familie \(\mathcal{S}_0\) de submodule
independente ale lui \(M\), maximală în raport cu relaţia de rafinare.

Fie \(N \in \mathcal{S}_0\).
Dacă \(N\) nu este coireductibil, există două submodule nenule
\(P,Q \leq N\) astfel încât \(P \cap Q = 0\).
Este clar că din \(\mathcal{S}_0\) se obţine o rafinare proprie a lui
\(\mathcal{S}_0\), contradicţie.
Rezultă că orice submodul din \(\mathcal{S}_0\) este coireductibil, deci
\(M\) conţine un submodul coireductibil.
Dacă \(L\) este un submodul nenul al lui \(M\), atunci şi \(L\) satisface
condiţia (2) şi, prin acelaşi raţionament, \(L\) conţine un submodul
coireductibil.

Fie acum
\[
S = \bigoplus_{i\in I} N_i
\]
o sumă directă maximală de submodule coireductibile din \(M\) şi
\(N\) un submodul nenul al lui \(M\).
Cum \(M\) satisface condiţia lanţurilor ascendente pentru sume directe,
rezultă că mulţimea \(I\) este finită.
Din Propoziţia~3.8 rezultă că \(S \cap N \neq 0\), ceea ce arată că
\(S \ess M\).
Prin definiţie, \(M\) are dimensiune coireductibilă finită.

\medskip
\emph{\((1) \Rightarrow (2)\).}
Fie
\[
S = \bigoplus_{i=1}^n N_i
\]
o sumă directă de submodule coireductibile astfel încât \(S \ess M\).
Dacă \(N\) este un submodul nenul al lui \(M\), atunci \(S \cap N \neq 0\),
iar din Lema~3.6 rezultă că \(N\) conţine un submodul coireductibil.

Fie \((L_j)_{j=1}^m\) o familie independentă de submodule ale lui \(M\).
Din paragraful precedent rezultă că fiecare \(L_j\) conţine un submodul
coireductibil \(P_j\).
Evident, familia \((P_j)_{j=1}^m\) este independentă.
Din Teorema~3.12 obţinem că \(m \leq n\).
Aceasta arată că orice familie independentă de submodule poate fi extinsă
numai până la o familie finită, ceea ce este echivalent cu condiţia
lanţurilor ascendente pentru sume directe.
\end{proof}

\begin{remark}
    Fie \(M\) un \(R\)-modul. Atunci:
\begin{enumerate}
  \item Dacă \(N \ess M\), atunci \(M\) are dimensiune coireductibilă
        finită dacă şi numai dacă \(N\) are dimensiune coireductibilă
        finită şi, în acest caz, \(\dim M = \dim N\).
  \item Dacă \(M = \bigoplus_{i=1}^n N_i\), atunci \(M\) are dimensiune
        coireductibilă finită dacă şi numai dacă fiecare \(N_i\) are
        dimensiune coireductibilă finită pentru orice \(i = 1,\dots,n\).
        În plus avem
        \[
          \dim M = \sum_{i=1}^n \dim N_i .
        \]
\end{enumerate}
\end{remark}
